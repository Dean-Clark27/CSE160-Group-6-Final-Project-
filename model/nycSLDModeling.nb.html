  # Loop through each factor and filter the test set

    # Get the levels the model learned

    # Filter test_fold to keep ONLY rows where the category exists in training
    # *Note that we lose a bit of data, but at least now the model works
    test_fold <- test_fold[test_fold[[col]] %in% train_levels, ]
  }
  
  # Predict
  p <- predict(m, test_fold)
  
  # Calculate Error in $ Dollars
  mse_fold <- mean((test_fold$price - p)^2)
  cv_mse_scores <- c(cv_mse_scores, mse_fold)
